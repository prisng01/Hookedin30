<script>
    function openEnvelope() {
        const flap = document.getElementById('flap');
        flap.style.transform = 'rotateX(180deg)';
        
        setTimeout(() => {
            document.getElementById('envelope').style.display = 'none';
            document.getElementById('card1').classList.add('active');
        }, 600);
    }

    function nextCard(cardNumber) {
        // Hide all cards
        document.querySelectorAll('.card').forEach(card => {
            card.classList.remove('active');
        });

        // Show requested card
        document.getElementById('card' + cardNumber).classList.add('active');
    }

    function selectRsvp(option) {
        const yesBox = document.getElementById('check-yes');
        const noBox = document.getElementById('check-no');

        if (option === 'yes') {
            yesBox.classList.add('checked');
            noBox.classList.remove('checked');
            createConfetti();
        } else {
            noBox.classList.add('checked');
            yesBox.classList.remove('checked');
        }
    }

    function createConfetti() {
        const container = document.getElementById('app');
        for (let i = 0; i < 30; i++) {
            const confetti = document.createElement('div');
            confetti.classList.add('confetti');
            confetti.style.left = Math.random() * 100 + '%';
            confetti.style.animationDelay = Math.random() * 1.5 + 's';
            confetti.style.backgroundColor = ['#a33835', '#d4af37', '#e28743', '#2565ae'][Math.floor(Math.random() * 4)];
            container.appendChild(confetti);

            setTimeout(() => confetti.remove(), 2500);
        }
    }
</script>
